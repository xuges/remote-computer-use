cmake_minimum_required(VERSION 3.16)
project(remote_computer_native C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_VISIBILITY_PRESET hidden)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
if(MINGW)
  include(CheckIncludeFile)
  check_include_file(threads.h RCU_HAS_C11_THREADS)
  if(NOT RCU_HAS_C11_THREADS)
    add_compile_definitions(__STDC_NO_THREADS__=1)
  endif()
endif()
option(RCU_BUILD_VNC "Build VNC adapter" ON)
option(RCU_BUILD_RDP "Build RDP adapter" ON)
if(RCU_BUILD_VNC)
  pkg_check_modules(GVNC REQUIRED IMPORTED_TARGET gvnc-1.0>=1.3)
  add_library(remote-computer-vnc SHARED common.c vnc.c)
  target_link_libraries(remote-computer-vnc PRIVATE PkgConfig::GVNC Threads::Threads)
  target_compile_options(remote-computer-vnc PRIVATE -Wall -Wextra -Wno-deprecated-declarations)
endif()
if(RCU_BUILD_RDP)
  pkg_check_modules(FREERDP REQUIRED IMPORTED_TARGET freerdp3>=3.25 freerdp-client3 winpr3)
  find_package(OpenSSL REQUIRED)
  add_library(remote-computer-rdp SHARED common.c rdp.c trust.c)
  target_link_libraries(remote-computer-rdp PRIVATE PkgConfig::FREERDP Threads::Threads OpenSSL::Crypto)
  target_compile_options(remote-computer-rdp PRIVATE -Wall -Wextra)
  if(APPLE)
    target_link_libraries(remote-computer-rdp PRIVATE "-framework Security" "-framework CoreFoundation")
  endif()
  if(WIN32)
    target_link_libraries(remote-computer-rdp PRIVATE ws2_32 crypt32)
  endif()
endif()

foreach(adapter remote-computer-vnc remote-computer-rdp)
  if(TARGET ${adapter})
    set_target_properties(${adapter} PROPERTIES PREFIX "lib")
    if(APPLE)
      set_target_properties(${adapter} PROPERTIES BUILD_RPATH "@loader_path" INSTALL_RPATH "@loader_path")
    endif()
  endif()
endforeach()
